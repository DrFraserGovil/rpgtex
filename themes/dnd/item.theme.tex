\RpgItemAddProperty{description}{\RpgItemDescription}{}
\RpgItemAddProperty{image}{\RpgItemImage}{}
\RpgItemAddProperty{type}{\RpgItemType}{}
\RpgItemAddProperty{rarity}{\RpgItemRarity}{}
\RpgItemAddProperty{requires-attunement}{\RpgAttunement}{none}

\keys_define:nn{rpg/rule/RpgItem}
{
	requires-attunement .value_required:n={false},
}

\ifdefined\tempwidth \else \newlength\tempwidth \fi


% \begin{noindent}
\RpgThemeItemFormat{
	\subsubsection{#1}
	\tl_if_blank:eF{\RpgItemRarity}
	{
		\tl_concat:NNN{\RpgItemRarity}{\RpgItemRarity}{~}
	}

	% BUG: this overwrites any description given by the user, so the description property is silently dropped in text mode (the card format shows it in preference to the rarity and type). Only build the line from rarity and type when no description was given, as the card format does.
	\tl_concat:NNN\RpgItemDescription{\RpgItemRarity}{\RpgItemType}

	\str_compare:eNeF{\RpgAttunement}{=}{none}
	{
		\tl_if_blank:eF{\RpgItemDescription}
		{
			\tl_concat:NNN\RpgItemDescription{\RpgItemDescription}{.~}
		}
		\tl_concat:NNN\RpgItemDescription{\RpgItemDescription}{Requires~attunement~\RpgAttunement{}}
	}
	\tl_if_blank:eF{\RpgItemDescription}
	{
		{{\small \it \RpgItemDescription}}

		\newline
		% \vspace{-0.4cm}
	}

	#2
}
