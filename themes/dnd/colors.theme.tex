%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Varying colour cycles
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%% Set the theme colour to increment at each part
\RpgThemeOnPartChange{
	\RpgAdvanceColors{}
}

\ExplSyntaxOn
\DeclareDocumentCommand\RpgAdvanceColors{}
{
	\__rpg_checkColorList:

	\seq_gpop_left:NN \__rpg_ColorCycle \l__tmpa_tl
	\RpgSetThemeColor{\l__tmpa_tl}
	\seq_gput_right:NV \__rpg_ColorCycle \l__tmpa_tl
}

\DeclareDocumentCommand\RpgThemeSetColorCycle{m}
{
	\seq_set_from_clist:Nn \__rpg_ColorCycle{#1}
	\RpgAdvanceColors{}
}
\DeclareDocumentCommand\RpgSetPHBColors{}
{
	\RpgThemeSetColorCycle{PhbLightGreen, PhbLightCyan, PhbMauve, PhbTan,DmgLavender,DmgCoral,DmgSlateGray,DmgPatina}
}
\DeclareDocumentCommand\RpgSetDMGColors{}
{
	\RpgThemeSetColorCycle{DmgLavender,DmgCoral,DmgSlateGray,DmgPatina,PhbLightGreen, PhbLightCyan, PhbMauve, PhbTan}
}
\cs_set_protected:Npn\__rpg_checkColorList:
{
	\seq_if_empty:NT \__rpg_ColorCycle
	{
		\msg_error:nnn{rpg}{empty-colors}{}
	}
}
\seq_clear_new:N \__rpg_ColorCycle
\msg_set:nnn {rpg}{empty-colors}{No~colors~in~color~cycle.~Call~RpgThemeSetColorCycle}

\ExplSyntaxOff
\RpgSetPHBColors{}
