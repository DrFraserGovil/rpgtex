%%%%%%%%%%%%%%%%%%%%%%%%%
%% Spell-Actions
%%%%%%%%%%%%%%%%%%%%%%%%%
\ExplSyntaxOn

\RpgThemeKeys{rpg/stat/spell}
{
	\RpgThemeTLKey{title}{\__dnd_statSpellTitle}{Spellcasting}
	\RpgThemeTLKey{bonus}{\__dnd_statSpellBonus}{0}
	\RpgThemeTLChoiceKey{modifier}{\__dnd_statSpellMod}{str,dex,con,int,wis,cha}{int}
	\RpgThemeTLMapKey{modifier}{\__dnd_statSpellMod}{STR,DEX,CON,INT,WIS,CHA}{str,dex,con,int,wis,cha}{INT}

	\RpgThemeBoolKey{innate}{\__dnd_statSpellInnate}
	\RpgThemeBoolKey*{no-intro}{\__dnd_statSpellUseHeader}
}

\DeclareDocumentEnvironment{RpgStatSpells}{ O{} +b}{
	\group_begin:
	\keys_set:nn{rpg/stat/spell}{#1}

	\bool_if:NT{\__dnd_statSpellUseHeader}
	{
		%% compute modifiers & sace DCs
		\__dnd_make_modifier:nnn{\__dnd_statSpellTotal}{\__dnd_statSpellMod}{\__dnd_statSpellBonus}
		\tl_set:Ne \__dnd_statSpellSave{\int_eval:n{8 + \__dnd_statSpellTotal}}
		\tl_set:Ne \__dnd_statNiceStat{\prop_item:Ne{\__dnd_StatNames}{\__dnd_statSpellMod}}

		\RpgStatAction{\__dnd_statSpellTitle} \ \RpgStatName{}\ may\ cast\ the\ following\ spells
		\bool_if:NT{\__dnd_statSpellInnate}
		{
			\ without\ requiring\ components
		}
		.\ They\ have\ \__dnd_statSpellTotal{}\ to\ hit,\ save\ DC\ \__dnd_statSpellSave

	}
	\providecommand{\perDay[2]}{\RpgStatSpellList[##1]{##2}}
	\providecommand{\atWill[1]}{\RpgStatSpellList{##1}}
	\providecommand{\level}{\RpgStatSpellSlots}

	\tl_if_empty:eF{#2}
	{
		\begin{description}[font = \normalfont \RpgFontStatBlockBody,
				labelsep = \__rpg_SpaceDim,
				itemindent=0.5em,
				noitemsep,
				topsep = 5pt plus 2pt minus 2pt]
			#2
		\end{description}
	}
}
{
	\group_end:
}

\DeclareDocumentCommand{\RpgStatSpellSlots}{o o m}
{
	\item[
	            \IfNoValueTF{#1}{Cantrips}{\RpgOrdinal{#1}\ level
		            \
		            (\IfNoValueTF{#2}{at\ will}{\RpgPlural{#2}{slot}})
	            }
	            :
	      ]
	\RpgEmphSpellString{#3}
}

\DeclareDocumentCommand{\RpgStatSpellList}{ o m}
{
	\item[
	            \IfValueTF{#1}{
		            \str_if_in:NnTF{#2}{,}
		            { #1/day\ each }
		            { #1/day }


	            }{
		            At\ will
	            }
	            :
	      ]
	\RpgEmphSpellString{#2}
}

%general wrapper for printing all spells in a list, with nice parsing of commans and whitespace
\DeclareDocumentCommand {\RpgEmphSpellString} {m}
{
	\group_begin:
	\seq_set_from_clist:Nn \l_tmpa_seq {#1}
	\seq_set_map:NNn \l_tmpb_seq \l_tmpa_seq { \exp_not:n { \it {##1} } }
	\seq_use:Nn \l_tmpb_seq { ,~ }
	\group_end:
}
\ExplSyntaxOff
