%% Formats the attack into the standard form:
%% Melee/Ranged (distance-in-ft): +x to hit. On hit [damage-calculation] damage, [optional plus] [optional or]. 

%\begin{noindent}
\DeclareDocumentCommand {\RpgStatAttack} {m O{}}
{
	\group_begin:
	\keys_set:nn { rpg / stat / attack } {#2}

	%%format the reach/range strings into the desired form
% BUG: the second line should act on \__dnd_attackRange, which never gets its 'ft'.
	\__dnd_append_if_missing:nn{\__dnd_attackReach}{ft}
	\__dnd_append_if_missing:nn{\__dnd_attackReach}{ft}

	% %Compute the to-hit modifier
	\__dnd_make_modifier:nnn{\__dnd_attackToHit}{\__dnd_attackStat}{\__dnd_attackBonus}

	%%Typeset the main attack
	\RpgStatAction{#1}~
	\str_case_e:nn {\__dnd_attackDistance}
	{
		{ melee }   {\textit{Melee\__dnd_attackType{}~(\__dnd_attackReach):}~}
% BUG: \l__range_tl does not exist, so ranged attacks show 'Ranged ():'. Use \__dnd_attackRange.
			{ ranged }  {\textit{Ranged\__dnd_attackType{}~(\l__range_tl):}~}
			{ both }    {\textit{Melee~(\__dnd_attackReach)\ or\ Ranged~(\__dnd_attackRange)\__dnd_attackType :}~}
	}
	{\__dnd_attackToHit}~to~hit.~

	\tl_set:Nn\l_tmpa_tl{\__dnd_on_hit_text:}
	\str_if_empty:NF {\l_tmpa_tl}
	{
	\textit { On~Hit : } ~ \l_tmpa_tl
	}
	\group_end:
}
% \end{noindent}

\cs_set_protected_nopar:Npn \__dnd_make_modifier:nnn#1#2#3
{
	\tl_set:Ne#1{\__dnd_format_signed_integer:N{\int_eval:n{\__dnd_statPB+#3+\use:c{__dnd_#2 Mod}}}}
}

%%% Construct the `on hit' text
\cs_set_protected:Npn \__dnd_on_hit_text:
{
	\tl_clear:N\l_tmpa_tl
	\str_if_empty:NF {\__dnd_attackDmg}
	{
		\tl_set:Ne\l_tmpa_tl{
			\RpgDice{\__dnd_attackDmg}\ \__dnd_attackDmgType\ damage
			\__dnd_if_or_dmg:
			\__dnd_if_plus_dmg:
			.\
		}
	}
	% `extra' is any special text that goes after the final damage; do not
	% include the final full stop.
	\str_if_empty:NF {\__dnd_attackExtra}
	{
		\tl_set:Ne\l_tmpa_tl{\l_tmpa_tl\__dnd_attackExtra .}
	}

	%%now place the output into the stream 	
	\l_tmpa_tl
}
%%Formatters

\cs_set_protected:Npn \__dnd_if_plus_dmg:
{
	\tl_if_empty:NF {\__dnd_attackPlusDmg}
	{ ~ plus\ \exp_args:Ne\RpgDice{\__dnd_attackPlusDmg}\ \__dnd_attackPlusType\ damage }
}

\cs_set_protected_nopar:Npn \__dnd_if_or_dmg:
{
	\tl_if_empty:NF {\__dnd_attackOrDmg}
	{
		, ~ or\ \exp_args:Ne\RpgDice{\__dnd_attackOrDmg}\ \__dnd_attackOrType\ damage\ \__dnd_attackOrWhen

% BUG: this tests \__dnd_attackOrDmg (always non-empty here), so the sentence ends ',.'. It presumably meant to add the comma only before a following plus-dmg clause.
		\tl_if_empty:NF {\__dnd_attackOrDmg}
		{,}
	}
}
