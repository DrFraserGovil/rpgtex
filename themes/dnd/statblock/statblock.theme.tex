\bool_set_false:N \__dnd_statIsBig



\NewDocumentEnvironment{RpgStat*}{O{} m O{} +b}
{
	\bool_set_true:N \__dnd_statIsBig
	\begin{RpgStat}[#1]{#2}[float-type=figure*,#3,twocolumn]
		#4
	\end{RpgStat}
}

% Float management
%%%%%%%%%%%%%%%%%%%

% \begin{noindent}
\cs_new_protected_nopar:Npn \__stat_begin:
{


	\tl_if_blank:eTF{\__dnd_statFloat}
	{
		\bool_if:NT \__dnd_statIsBig
		{
			\str_compare:eNeT{\__rpg_PageColumns}{=}{twocolumn}
			{
				\tl_set:Ne{\__dnd_statFloat}{-}	
				\tl_set:Ne{\__dnd_statFloatType}{strip}
				\begin{strip}
			}
		}
	}
	{
		\edef\floatbegin{\noexpand\begin{\__dnd_statFloatType}[\__dnd_statFloat]}%
		\floatbegin
	}

	\begin{RpgStatBox}[\__dnd_statBoxOptions]

		\bool_if:NT\__dnd_statTwocol{ \begin{multicols}{2} }
}

\cs_new_protected_nopar:Npn\__stat_end:
{
		\bool_if:NT\__dnd_statTwocol { \end{multicols} } 
	\end{RpgStatBox}
	 
	\tl_if_blank:eF{\__dnd_statFloat}
	{
		\expandafter\end\expandafter{\__dnd_statFloatType}
	}
}
