\RpgIfSections{
	\renewcommand{\thepart}{\arabic{part}}
	%% Contents formatting is only set in the preamble: after \begin{document}, titletoc writes \titlecontents into the .toc file,
	%% where it would refer to this theme's colours and fonts before the theme has been loaded on the next run
	\RpgIfPreamble{
		\titlecontents{part}
		[0pt] % left
		{ \addvspace{.3cm} \titleline{ \color{black} \titlerule[1pt] } \addvspace{.1cm }  \RpgFontTocPart \centering } % above-code
		{Part\ \thecontentslabel\\*} % numbered-entry-format
		{\centering} % numberless-entry-format
		{} % filler-page-format
		[  \addvspace{.1cm } \titleline{ \color{black} \titlerule[1pt] } ] % below-code
	}
}

\ExplSyntaxOn
\DeclareDocumentCommand\RpgIfColoursEqual{m m +m +m}
{
	% Extract and convert Color 1 to RGB
	\extractcolorspecs {#1} \l_tmpa_tl \l_tmpb_tl
	\convertcolorspec \l_tmpa_tl \l_tmpb_tl { rgb } \l_tmpa_tl

	% Extract and convert Color 2 to RGB
	\extractcolorspecs {#2} \l_tmpc_tl \l_tmpd_tl
	\convertcolorspec \l_tmpc_tl \l_tmpd_tl { rgb } \l_tmpb_tl

	% Compare normalized RGB component strings
	\str_if_eq:NNTF \l_tmpa_tl \l_tmpb_tl
	{#3}
	{#4}
}
