%%%%%%%%%%%%%%%%%%%%%%%%%%
% Section manipulation
%%%%%%%%%%%%%%%%%%%%%%%%%%

%% These are section-related commands, not about the formatting of sections (that belongs in the Themes)


\def\RpgPartName{~}
\def\RpgChapterName{~}
\def\RpgSectionName{~}
\def\RpgSubsectionName{~}
\def\RpgSubsubsectionName{~}

\ExplSyntaxOn
\msg_new:nnn{rpg}{bad-section}{#1~is~not~a~valid~section-level~name}
\NewDocumentCommand{\RpgGetName}{m}
{
	\str_case:enF{#1}
	{
		{part} {\RpgPartName}
			{chapter} {\RpgChapterName}
			{section} {\RpgSectionName}
			{subsection} {\RpgSubsectionName}
			{subsubsection} {\RpgSubsubsectionName}
	}
	{
		\msg_error:nnn{rpg}{bad-section}{#1}
	}
}

% \begin{noindent}
\NewDocumentCommand\RpgFake{m m}
{
	\phantomsection
	\str_case:enTF{#1}
	{
		{part}{\def\RpgPartName{#2}}
		{chapter}{\chaptermark{#2}}
		{section}{\sectionmark{#2}}
		{subsection}{\subsectionmark{#2}}
		{subsubsection}{\subsubsectionmark{#2}}
	}
	{
		\addcontentsline{toc}{#1}{#2}
	}
	{
		\msg_error:nnn{rpg}{bad-section}{#1}
	}
}

%% Overload '[x]mark' commands to store the names in the names as well 

\let\__rpg_chaptermark\chaptermark
\let\__rpg_sectionmark\sectionmark
\let\__rpg_subsectionmark\subsectionmark
\let\__rpg_subsubsectionmark\subsubsectionmark

\RpgIfSections{
	\setcounter{secnumdepth}{0}
	\def\chaptermark#1{\__rpg_chaptermark{#1}\def\RpgChapterName{#1}}
	\def\sectionmark#1{\__rpg_sectionmark{#1}\def\RpgSectionName{#1}}
	\def\subsectionmark#1{\__rpg_subsectionmark{#1}\def\RpgSubsectionName{#1}}
	\def\subsubsectionmark#1{\__rpg_subsubsectionmark{#1}\def\RpgSubsubsectionName{#1}}
}
% \end{noindent}
