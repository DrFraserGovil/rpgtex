%%%%%% Theme switching
\ExplSyntaxOn


%% The function used to switch themes. Immediately loads in the new overloads, overloading any previous definitions
%% I
% BUG:  themes cannot be loaded twice in one document.
% Theme files define commands with \NewDocumentCommand/\newcommand/\cs_new/\msg_new (e.g. \ConditionalRagged in default/callouts.theme.tex, most of dnd/colors, spell and statblock/*), so any mid-document switch that reloads a theme (including the \RpgSetTheme{default} inside dnd.cfg and scifi.cfg) raises "already defined" errors. Planned fix: switch theme definitions to \DeclareDocumentCommand/\cs_set (so later themes can override them; \Provide... only where the first definition should win), and guard variables and messages with \..._if_exist or \msg_set.
\NewDocumentCommand\RpgSetTheme { m }
{

	%%%first try looking in the package themes
	\def\temp@themePath{\__rpg_ThemePath/#1/#1.cfg}
	\file_if_exist:nTF \temp@themePath
	{
		\input{\tl_use:N \temp@themePath}
	}
	{
		%%% Then try looking in my *original* path
		\file_if_exist:nTF{\RpgPackagePath/themes/#1/#1.cfg}
		{
			\input{\RpgPackagePath/themes/#1/#1.cfg}
		}
		{
			%%%if not found then try reading it as a direct reference
			\file_if_exist:nTF{#1}
			{
				\input{#1}
			}
			{
				\msg_error:nnxx { rpg } { theme_missing } { #1 } { \temp@themePath }
			}
		}
	}

}

\msg_new:nnn { rpg } { theme_missing } {Could~not~locate~theme~#1~directly,~and~is~not~on~theme~path(#2)}

\NewDocumentCommand\RpgSetThemePath { m }
{
	\tl_set:Nn \__rpg_ThemePath {#1}
}
