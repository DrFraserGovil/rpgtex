%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Conditionals
%% Runs code depending on the active package options, or on the current position in the document
%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\ExplSyntaxOn

%% Runs #1 if the font option is active, and #2 otherwise
\NewDocumentCommand\RpgIfFontTF{+m +m}
{
	\bool_if:NTF \__rpg_UseFontOption {#1} {#2}
}
\NewDocumentCommand\RpgIfFont{+m}
{
	\RpgIfFontTF {#1} {}
}

%% Runs #1 in the preamble, and #2 in the document body
%% \@nodocument is \relax only after \begin{document}. It is called by name, so this works even when @ is not a letter
\NewDocumentCommand\RpgIfPreambleTF{+m +m}
{
	\cs_if_eq:cNTF {@nodocument} \scan_stop: {#2} {#1}
}
\NewDocumentCommand\RpgIfPreamble{+m}
{
	\RpgIfPreambleTF {#1} {}
}
