%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Secrets
%%%%%%%%%%%%%%%%%%%%%%%%%%

\RpgNewSwitch{ShowSecrets}[off]

\RpgThemeKeys{rpg/secret}
{
	\RpgThemeTLKey{color}{\__rpg_secretColor}{SecretColor}
	\RpgThemeTLKey{label-public}{\__rpg_secretLabelPublic}{Normal~text}
	\RpgThemeTLKey{label-secret}{\__rpg_secretLabelSecret}{Hidden~text}
	\RpgThemeTLKey{switch}{\__rpg_secretSwitch}{ShowSecrets}
}

% \begin{noindent}

%% RpgSecret allows text to render differently depending on switch
%% Arguments:	#1 keyvals (or tcbox fallthroughs)
%%				#2 `Visible text' which is displayed when the Switch is off
%%				#3 `Hidden text' which is displayed when the Switch is on	
\NewDocumentEnvironment{RpgSecret}{O{} +G{} +b}
{
	\keys_set_known:nnN{rpg/secret}{#1}\__rpg_secretTmp
	 
	\exp_args:Ne\RpgIfSwitch{\__rpg_secretSwitch}
	{
		\__rpg_SecretVisible:nnn{#2}{\__rpg_secretTmp}
	}
	{
		\__rpg_SecretHidden:nnn{#2}{#3}{\__rpg_secretTmp}
	}
} { }
% \end{noindent}


%%%%%%%%%%%%%%%
%% Design
%%%%%%%%%%%%%%%
\tl_new:N \__rpg_secretTmp

%% Theme setters for the visible and hidden modes, controlling how they appear
%% Args for both the same (though HIdden doesn't need to use #3, normally!)
%%	#1	The `Visible Text' (possibly empty)
%%	#2	The 'Secret Text'
%%	#3	Fallthrough arguments (usually expected to be tcbox commands)
\NewDocumentCommand\RpgThemeSetSecretVisible{ +m }
{
	\cs_set:Nn \__rpg_SecretVisible:nnn{#1}
}
\NewDocumentCommand\RpgThemeSetSecretHidden{ +m }
{
	\cs_set:Nn \__rpg_SecretHidden:nnn{#1}
}

\cs_new_protected:Npn \__rpg_SecretVisible:nnn#1#2#3{ }
\cs_new_protected:Npn \__rpg_SecretHidden:nnn#1#2#3{ }

\tcbset{
	rpgsecret/.style={
			standard~jigsaw, %for transparency
			opacityback=0,
			%%set dimensions -- skinny and lightweight so doesn't interrupt text too much
			boxrule=1pt,
			left=2pt,
			right=2pt,
			top=2pt,
			bottom=2pt,
			sidebyside~gap=0.4cm
		}

}

%%% Use the above commands to set the default
\RpgThemeSetSecretHidden
{
	#1
}
\RpgThemeSetSecretVisible
{
	\begin{tcolorbox}[rpgsecret,colframe=\__rpg_secretColor,#3]
		{\RpgFontSecretTitle \__rpg_secretLabelSecret}\par
		#2
		\tl_if_blank:eF{#1}
		{
			\tcblower
			{\RpgFontSecretTitle \__rpg_secretLabelPublic}\par
			#1
		}
	\end{tcolorbox}

}
