%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Conditional multi-envs
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\ExplSyntaxOn

\NewDocumentEnvironment{RpgIfEnv}{m O{} m O{} m}
{
	\group_begin:
	\RpgIfSwitch{#1}
	{
		\tl_set:Ne \__rpg_env{#3}
		\__rpg_beginEnv:nn{#3}{#2}
	}
	{
		\tl_set:Ne \__rpg_env{#5}
		\__rpg_beginEnv:nn{#5}{#4}
	}
}
{
	\exp_args:Ne\end{\__rpg_env}
	\group_end:
}

%% Protect against optional argument expansion by ensuring opts only passed if present
%% This means we don't need to worry about O{} vs o downstream
\cs_new_protected:Npn \__rpg_beginEnv:nn#1#2
{
	\tl_if_empty:eTF{#2}
	{
		\begin{#1}
	}
	{
		\begin{#1}[#2]
	}
}
