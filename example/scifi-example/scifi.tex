\documentclass[theme=scifi]{rpghandout}

\title{Mission Briefing}
\subtitle{Salvage Operation: The Halcyon Drift}
\author{Fleet Command}

\begin{document}
\maketitle

\section{Situation}

\RpgDropCap{T}{he freighter} \textit{Halcyon Drift} went silent eleven days ago, while hauling medical supplies to the Carrow colonies. A long-range scan shows her drifting, intact, with her reactor still running.

\begin{RpgNarration}
	Your shuttle docks with a dull clang. Beyond the airlock, the corridor lights flicker in a slow, regular pattern, as if something is signalling.
\end{RpgNarration}

\section{Objectives}

\begin{RpgTable}[filigree, title={Mission Objectives}]{lXl}
	Priority & Objective & Reward
	\\ Primary & Recover the ship's flight recorder. & 5,000 cr
	\\ Secondary & Secure the cargo of medical supplies. & 2,000 cr
	\\ Optional & Learn what happened to the crew. & ---
\end{RpgTable}

\begin{RpgSidebar}{Hazards}
	Boarding parties should expect vacuum in the cargo bay, and a damaged reactor. Each hour spent in the engine room deals \RpgDice{1d6} radiation damage.
\end{RpgSidebar}

\begin{RpgTip}{Fleet Command Advises}
	Do not open the medical bay until the flight recorder has been recovered.
\end{RpgTip}
