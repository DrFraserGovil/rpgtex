\NewDocumentCommand\MarginBlock{+m O{black} O{} +m O{white} +m}
{
	\par \vspace{0.3cm}

	{
		\hspace{-\dimexpr\reducer\relax} %
		\begin{minipage}{0.922\paperwidth}
			\colorbox{#5}{\makebox[0.99\paperwidth][l]{
					\parbox[t]{\macrowidth}{
						\mbox{} \raggedleft \color{#2} \maxsizebox{\macrowidth}{!}{\tt #1} }\hspace{\macrosep}
					\parbox[t]{\dimexpr0.94\linewidth - \macrowidth-\macrosep\relax}{ \textcolor{#2}{#4}
					}
				}
			}
		\end{minipage}
	}

	\par
	\tl_set:Ne\l_tmpa_tl{#3}
	\tl_if_blank:oTF{\l_tmpa_tl}
	{
		\begin{adjustwidth}{2.3cm}{0pt}
			#6
		\end{adjustwidth}

	}
	{
		\hspace{-\dimexpr\reducer\relax}
		\hspace{\fboxsep}\begin{minipage}[t]{\macrowidth}
			\mbox{} \raggedleft \color{#2} \maxsizebox{\macrowidth}{!}{\tt #3 }
		\end{minipage}%
		\hspace{\macrosep}%
		\begin{minipage}[t]{\dimexpr\linewidth-\macrowidth+0.5\reducer\relax}

			#6
		\end{minipage}
	}

}
