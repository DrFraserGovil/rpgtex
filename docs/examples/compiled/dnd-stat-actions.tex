\begin{ExampleBlock}[RpgStat, section, trait, action, reaction][listing side text]{Sections, Traits \& Reactions}
\begin{RpgStat}{Guard}[ac=16 (chain shirt), hp=\RpgDice{3d8+3}, str=13, con=12]
    \section{Traits}
        \trait{Watchful} The guard can't be surprised while conscious.
    \section{Actions}
        \action{Halberd Sweep} The guard attacks each creature within 10ft.
    \section{Reactions}
        \reaction{Parry}{When hit by a melee attack} The guard adds 2 to its AC.
\end{RpgStat}


\end{ExampleBlock}
