%% Shows the cover and part pages of a theme, rendered by the mockup document examples/pages/page-<theme>.tex (built by makepages)
\NewDocumentCommand{\ThemePages}{m}{
	\begin{center}
		\begin{tabular}{cc}
			\fbox{\includegraphics[page=1,width=0.4\linewidth]{examples/pages/page-#1.pdf}}
			                  &
			\fbox{\includegraphics[page=2,width=0.4\linewidth]{examples/pages/page-#1.pdf}}
			\\
			\small Cover~page & \small Part~page
		\end{tabular}
	\end{center}
}

\prop_new:N \table_names
\NewDocumentCommand{\FontTableElement}{m m}
{
	%%Add entries into the registry
	%%extracts the current values assigned to the variables
	\prop_put:Nen{\table_names}{#2}{\verb|#1| & \texttt{\tl_to_str:c{__rpg_font_#2_family}}& \texttt{\tl_to_str:c{__rpg_font_#2_style}} & #1 Lorum~ipsum}

}
\NewDocumentCommand{\InsertFontTable}{m}{
	% \keys_set:nn{fonttable}{#1}

	\begin{RpgTable}[breakable]{lllX}
		Element & Font Family & Font Style & Example
		\prop_map_inline:Nn\table_names{
			\\ ##2
		}
	\end{RpgTable}
}
